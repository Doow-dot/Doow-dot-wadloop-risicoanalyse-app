// ---------- Water (Rijkswaterstaat) ---------- const STATIONS = { lauwersoog: ["lauwersoog.waddenzee", "lauwersoog"], nes: ["ameland.nes", "nes"], harlingen: ["harlingen.waddenzee", "harlingen"], eemshaven: ["eemshaven", "eemshaven.waddenzee"], }; const RWS = "https://ddapi20-waterwebservices.rijkswaterstaat.nl/ONLINEWAARNEMINGENSERVICES/OphalenWaarnemingen"; const iso = (d) => d.toISOString().replace("Z", "+00:00"); function lokaal(d) { const opties = { timeZone: "Europe/Amsterdam" }; } async function haalVerwachting(code) { const nu = new Date(); const van = new Date(nu.getTime() - 60 * 60 * 1000); const tot = new Date(nu.getTime() + 48 * 60 * 60 * 1000); const res = await fetch(RWS, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ AquoPlusWaarnemingMetadata: { AquoMetadata: { Compartiment: { Code: "OW" }, Eenheid: { Code: "cm" }, Grootheid: { Code: "WATHTE" }, ProcesType: "verwachting", }, }, Locatie: { Code: code }, Periode: { Begindatumtijd: iso(van), Einddatumtijd: iso(tot) }, }), }); } } return punten.sort((a, b) => a.t - b.t); } function vindLaagwaters(punten) { const nu = Date.now(); const uit = []; for (let i = 6; i < punten.length - 6; i++) { if (punten[i].t.getTime() < nu) continue; } catch (e) { return { station, beschikbaar: false, fout: String(e.message || e), opgehaald: new Date().toISOString() }; } } // ---------- Weer (KNMI HARMONIE-AROME via Open-Meteo) ---------- const OM_VARS = "temperature_2m,apparent_temperature,wind_speed_10m,wind_direction_10m,wind_gusts_10m,precipitation,snowfall,visibility,weather_code"; const MODELLEN = ["knmi_harmonie_arome_netherlands", "knmi_harmonie_arome_europe"]; const KOMPAS = ["N", "NO", "O", "ZO", "Z", "ZW", "W", "NW"]; const kompas = (g) => KOMPAS[Math.round(g / 45) % 8]; const isNum = (v) => typeof v === "number" && isFinite(v); const r1 = (x) => Math.round(x * 10) / 10; function bft(ms) { const grens = [0.3, 1.6, 3.4, 5.5, 8.0, 10.8, 13.9, 17.2, 20.8, 24.5, 28.5, 32.7]; let b = 0; while (b < grens.length && ms >= grens[b]) b++; return b; } async function haalWeer(vanMs, totMs) { const start = Math.floor(vanMs / 3600000) * 3600000; const einde = Math.ceil(totMs / 3600000) * 3600000; const verwacht = (einde - start) / 3600000 + 1; const wind = idx.map((i) => h.wind_speed_10m[i]); const iMax = idx[wind.indexOf(Math.max(...wind))]; const windMax = h.wind_speed_10m[iMax]; const graden = h.wind_direction_10m[iMax]; const vlaag = num("wind_gusts_10m"); const temp = num("temperature_2m"); const gev = num("apparent_temperature"); const zicht = num("visibility"); const codes = num("weather_code"); return { beschikbaar: true, locatie: LOCATIE.naam, model, bron: "KNMI HARMONIE-AROME via Open-Meteo.com", opgehaald: new Date().toISOString(), venster: { van: new Date(start).toISOString(), tot: new Date(einde).toISOString() }, samenvatting: { windMaxMs: r1(windMax), windMaxBft: bft(windMax), windstootMaxMs: vlaag.length ? r1(Math.max(...vlaag)) : null, windrichtingGraden: Math.round(graden), windrichting: kompas(graden), tempMin: r1(Math.min(...temp)), tempMax: r1(Math.max(...temp)), tempGem: r1(temp.reduce((a, b) => a + b, 0) / temp.length), gevoelMin: gev.length ? r1(Math.min(...gev)) : null, gevoelMax: gev.length ? r1(Math.max(...gev)) : null, zichtMinM: zicht.length ? Math.round(Math.min(...zicht)) : null, neerslagMm: r1(som("precipitation")), sneeuwCm: r1(som("snowfall")), ijs: codes.some((c) => [56, 57, 66, 67].includes(c)), }, uren: idx.map((i) => ({ tijdISO: new Date(tijden[i]).toISOString(), windMs: r1(h.wind_speed_10m[i]), richting: Math.round(h.wind_direction_10m[i]), temp: r1(h.temperature_2m[i]), zichtM: h.visibility && isNum(h.visibility[i]) ? Math.round(h.visibility[i]) : null, neerslagMm: r1(h.precipitation[i]), })), }; } return { beschikbaar: false, reden, verwachtingTot: laatste ? new Date(laatste).toISOString() : null }; } // ---------- Waarschuwingen (KNMI public weather warnings) ---------- const KDP = "https://api.dataplatform.knmi.nl/open-data/v1/datasets/KNMI-public-weather-warnings/versions/1.0/files"; async function laatsteSnapshot(sleutel) { const lijst = await fetch(KDP + "?maxKeys=20&sorting=desc&orderBy=filename", { headers: { Authorization: sleutel } }); if (!lijst.ok) throw new Error("KNMI bestandslijst gaf status " + lijst.status); const l = await lijst.json(); const namen = (l.files || []).map((f) => f.filename).filter((n) => /\.json$/i.test(n)).slice(0, 5); if (!namen.length) throw new Error("geen JSON-bestanden gevonden in de KNMI-dataset"); for (const naam of namen) { const u = await fetch(KDP + "/" + encodeURIComponent(naam) + "/url", { headers: { Authorization: sleutel } }); if (!u.ok) throw new Error("KNMI download-link gaf status " + u.status); const link = (await u.json()).temporaryDownloadUrl; const f = await fetch(link); if (!f.ok) throw new Error("KNMI bestand gaf status " + f.status); const snap = await f.json(); if (snap.metadata && snap.metadata.publication_type === "test") continue; return { naam, snap }; } throw new Error("alleen testbestanden gevonden"); } function ringBevat(ring, lon, lat) { let binnen = false; for (let i = 0, j = ring.length - 1; i < ring.length; j = i++) { const [xi, yi] = ring[i], [xj, yj] = ring[j]; if (yi > lat !== yj > lat && lon < ((xj - xi) * (lat - yi)) / (yj - yi) + xi) binnen = !binnen; } return binnen; } const polygonen = (a) => (!a ? [] : a.type === "MultiPolygon" ? a.coordinates : [a.coordinates]); const bevatPunt = (a, lon, lat) => polygonen(a).some((p) => p && p[0] && ringBevat(p[0], lon, lat)); function raaktRegio(a) { return polygonen(a).some((p) => { if (!p || !p[0]) return false; }); } // Zet de KNMI-verschijnselnaam om naar de keuzelijst in het formulier. function soortVoorFormulier(ph) { const s = String(ph || "").toLowerCase().replace(/[^a-z]/g, ""); if (/fog|mist/.test(s)) return "Mist"; } const ERNST = { red: 3, orange: 2, yellow: 1 }; async function waarschuwingenBody(env, vanMs, totMs) { if (!env.KNMI_OPEN_DATA_KEY) { return { beschikbaar: false, fout: "De KNMI-sleutel ontbreekt in de Worker (geheim KNMI_OPEN_DATA_KEY)." }; } } // ---------- Aanvraag afhandelen ---------- async function metCache(request, ctx, ttl, cors, bouw) { const cache = caches.default; const sleutel = new Request(request.url); const gecacht = await cache.match(sleutel);
